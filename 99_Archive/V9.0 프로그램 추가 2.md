```python
# modules/skill_spector.py
# Hermes3 v9.4.2 Production - Lightweight SkillSpector Telemetry Engine
# Cleanroom Extraction of NVIDIA's Trajectory Profiling Philosophy

import os
import time
import sqlite3
import logging
from dataclasses import dataclass

logger = logging.getLogger('SkillSpector')

@dataclass
class SkillProfile:
    action: str
    query: str
    start_time: float
    end_time: float = 0.0
    char_count: int = 0
    is_success: bool = True
    error_message: str = ""

    @property
    def latency(self) -> float:
        return self.end_time - self.start_time


class LightweightSkillSpector:
    """[NVIDIA SkillSpector 사상 이식] 스킬 실행 비용과 무결성을 계량화하는 초경량 프로파일러"""
    
    def __init__(self, db_path: str = "~/.hermes/runtime/dreaming_workbench.db"):
        self.db_path = os.path.expanduser(db_path)
        self._init_telemetry_table()

    def _init_telemetry_table(self):
        conn = sqlite3.connect(self.db_path, timeout=10.0)
        try:
            conn.execute("PRAGMA journal_mode=WAL;")
            # 스킬 가성비(ROI) 분석용 원격측정 테이블 구축
            conn.execute("""
                CREATE TABLE IF NOT EXISTS skill_telemetry (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    action TEXT,
                    query TEXT,
                    latency REAL,
                    char_count INTEGER,
                    is_success INTEGER,
                    recorded_at REAL
                );
            """)
            conn.commit()
        finally:
            conn.close()

    def profile_start(self, action: str, query: str) -> SkillProfile:
        """도구 가동 직전 시간선 박제"""
        return SkillProfile(action=action, query=query, start_time=time.time())

    def profile_end(self, profile: SkillProfile, result_text: str, is_success: bool = True, err_msg: str = ""):
        """도구 종료 후 소모 자원 및 Latency 계량화 저장"""
        profile.end_time = time.time()
        profile.char_count = len(result_text)
        profile.is_success = is_success
        profile.error_message = err_msg

        # 공통 SQLite 백엔드에 즉시 영속화 (인프라 부하 0B 메모리 고수)
        conn = sqlite3.connect(self.db_path, timeout=10.0)
        try:
            conn.execute(
                "INSERT INTO skill_telemetry (action, query, latency, char_count, is_success, recorded_at) VALUES (?, ?, ?, ?, ?, ?)",
                (profile.action, profile.query, profile.latency, profile.char_count, 1 if profile.is_success else 0, time.time())
            )
            conn.commit()
            logger.info(f"📊 [v9.4.2 스펙터] 스킬 검사 완결 -> {profile.action} | Latency: {profile.latency:.3f}s | Size: {profile.char_count}자")
        finally:
            conn.close()

```

```python
# modules/cyber_auditor.py
# Hermes3 v9.5 Production - Isolated Cybersecurity Static Auditor Module
# Implementation of Anthropic's Cybersecurity Tooling Blueprint

import os
import re
import json
import logging
from typing import Dict, Any

logger = logging.getLogger('CyberAuditor')

class SecureCodeAuditor:
    """[Anthropic Cyber Skills 이식] 소스코드 및 스크립트 취약점을 정적으로 감사하는 격리형 보안 엔진"""
    
    def __init__(self):
        # OWASP Top 10 및 주요 CWE 취약점 패턴 사전 정의 (결정론적 가드)
        self.vulnerability_patterns = {
            "Hardcoded Key": r"(secret_key|password|passwd|api_key)\s*=\s*['\"][a-zA-Z0-9]{16,128}['\"]",
            "Command Injection Risk": r"(os\.system|subprocess\.Popen|eval|exec)\(.*?\+.*?\)",
            "Path Traversal Risk": r"open\(.*?(\.\./|\.\.\\).*?\)"
        }

    def audit_source_code(self, raw_code: str) -> Dict[str, Any]:
        """로컬 가용 자원만으로 코드 내 취약점 패턴을 0.001초 만에 검독하는 순수 분석 함수"""
        detected_issues = []
        
        for vuln_name, pattern in self.vulnerability_patterns.items():
            matches = re.finditer(pattern, raw_code, re.IGNORECASE)
            for match in matches:
                detected_issues.append({
                    "vulnerability": vuln_name,
                    "matched_text": match.group(0)[:100],
                    "risk_level": "CRITICAL" if vuln_name != "Hardcoded Key" else "HIGH"
                })
```

```python
class IncidentInvestigator:
    """[Tracer-Cloud/OpenSRE 사상 이식] 블랙박스 추론을 거부하고 지표/로그를 결정론적으로 연쇄 추적하는 SRE 진단 가드"""
    
    def __init__(self, log_dir: str = "~/.hermes/runtime/logs"):
        self.log_dir = os.path.expanduser(log_dir)
        os.makedirs(self.log_dir, exist_ok=True)

    def correlate_alert_signals(self, raw_alert: Dict[str, Any]) -> Dict[str, Any]:
        """
        알림 페이로드를 받아 연관된 로그 버퍼와 시스템 위험도를 동적으로 추출 (Factor 4)
        """
        alert_source = raw_alert.get("source", "unknown")
        message = raw_alert.get("message", "")

        error_keywords = ["OOM", "502 Bad Gateway", "Connection Refused", "Timeout"]
        detected_triggers = [kw for kw in error_keywords if kw.lower() in message.lower()]
        
        # 🛡️ 가드 2: 가상 타겟 서비스 및 영향 범위(Blast Radius) 가이드 생성
        service_target = re.search(r"service:\s*([a-zA-Z0-9_-]+)", message, re.IGNORECASE)
        service_name = service_target.group(1) if service_target else "global_kernel"
        
        # 에이전트가 소화할 수 있도록 증거 기반(Evidence-backed) 컨텍스트로 정제 리턴
        return {
            "incident_detected": len(detected_triggers) > 0,
            "target_service": service_name,
            "matched_triggers": detected_triggers,
            "evidence_snapshot": f"Alert Source: {alert_source} | Timestamp: {time.time()} | Payload: {message[:200]}"
        }
```
